\section*{Gabarito — Binômio e Multinômio de Newton}

\begin{enumerate}

% -------------------------------------------------------
\item \textbf{(Expansão Direta — Binômio)} Expanda as expressões abaixo utilizando o Teorema do Binômio de Newton e simplifique os coeficientes:

\begin{solucao}
\begin{enumerate}[label=\alph*)]
    \item $(x + 2)^5 = \binom{5}{0}x^5 + \binom{5}{1}x^4(2)^1 + \binom{5}{2}x^3(2)^2 + \binom{5}{3}x^2(2)^3 + \binom{5}{4}x(2)^4 + \binom{5}{5}(2)^5$ \\
    $= 1x^5 + 5x^4(2) + 10x^3(4) + 10x^2(8) + 5x(16) + 32$ \\
    $= \mathbf{x^5 + 10x^4 + 40x^3 + 80x^2 + 80x + 32}$

    \item $(3a - b)^4 = \binom{4}{0}(3a)^4 + \binom{4}{1}(3a)^3(-b)^1 + \binom{4}{2}(3a)^2(-b)^2 + \binom{4}{3}(3a)^1(-b)^3 + \binom{4}{4}(-b)^4$ \\
    $= 1(81a^4) + 4(27a^3)(-b) + 6(9a^2)(b^2) + 4(3a)(-b^3) + 1b^4$ \\
    $= \mathbf{81a^4 - 108a^3b + 54a^2b^2 - 12ab^3 + b^4}$

    \item $\left(1 - \dfrac{x}{2}\right)^6 = \binom{6}{0}(1)^6 + \binom{6}{1}(1)^5(-\frac{x}{2})^1 + \binom{6}{2}(1)^4(-\frac{x}{2})^2 + \binom{6}{3}(1)^3(-\frac{x}{2})^3 + \binom{6}{4}(1)^2(-\frac{x}{2})^4 + \binom{6}{5}(1)^1(-\frac{x}{2})^5 + \binom{6}{6}(-\frac{x}{2})^6$ \\
    $= 1 - 6(\frac{x}{2}) + 15(\frac{x^2}{4}) - 20(\frac{x^3}{8}) + 15(\frac{x^4}{16}) - 6(\frac{x^5}{32}) + \frac{x^6}{64}$ \\
    $= \mathbf{1 - 3x + \frac{15}{4}x^2 - \frac{5}{2}x^3 + \frac{15}{16}x^4 - \frac{3}{16}x^5 + \frac{1}{64}x^6}$
\end{enumerate}
\end{solucao}

% -------------------------------------------------------
\item \textbf{(Termo Geral)} Determine o que se pede em cada item:

\begin{solucao}
\begin{enumerate}[label=\alph*)]
    \item \textbf{5º termo de $(2x + y)^8$:} \\
    $T_{k+1} = \binom{n}{k}a^{n-k}b^k$. Para o 5º termo, $k = 4$. \\
    $T_5 = \binom{8}{4}(2x)^{8-4}(y)^4 = 70 \cdot (2x)^4 \cdot y^4 = 70 \cdot 16x^4 \cdot y^4 = \mathbf{1120x^4y^4}$

    \item \textbf{Coeficiente de $x^3$ em $(1 + 3x)^7$:} \\
    $T_{k+1} = \binom{7}{k}(1)^{7-k}(3x)^k = \binom{7}{k} 3^k x^k$. \\
    Para $x^3$, temos $k = 3$. \\
    $T_4 = \binom{7}{3} 3^3 x^3 = 35 \cdot 27 x^3 = 945 x^3$. \\
    O coeficiente é \textbf{945}.

    \item \textbf{Termo independente de $x$ em $(x^2 - 1/x)^9$:} \\
    $T_{k+1} = \binom{9}{k}(x^2)^{9-k}(-x^{-1})^k = \binom{9}{k} x^{18-2k} (-1)^k x^{-k} = \binom{9}{k} (-1)^k x^{18-3k}$. \\
    Para o termo independente, o expoente de $x$ deve ser zero: \\
    $18 - 3k = 0 \implies 3k = 18 \implies k = 6$. \\
    $T_7 = \binom{9}{6}(-1)^6 = \frac{9 \cdot 8 \cdot 7}{3 \cdot 2 \cdot 1} \cdot 1 = 84$. \\
    O termo independente é \textbf{84}.
\end{enumerate}
\end{solucao}

% -------------------------------------------------------
\item \textbf{(Identidade de Pascal)} Verifique a expansão de $(a+b)^6$ utilizando o triângulo:

\begin{solucao}
Os coeficientes da linha 6 do Triângulo de Pascal são: 1, 6, 15, 20, 15, 6, 1. \\
Portanto, a expansão é: \\
$\mathbf{(a+b)^6 = a^6 + 6a^5b + 15a^4b^2 + 20a^3b^3 + 15a^2b^4 + 6ab^5 + b^6}$
\end{solucao}

% -------------------------------------------------------
\item \textbf{(Multinômio de Newton — Coeficientes)} Determine os coeficientes pedidos:

\begin{solucao}
A fórmula do coeficiente multinomial é $C = \dfrac{n!}{k_1!k_2!\dots k_m!}$.
\begin{enumerate}[label=\alph*)]
    \item \textbf{Coeficiente de $x^2 y^2 z$ em $(x + y + z)^5$:} \\
    $n=5, k_1=2, k_2=2, k_3=1$. \\
    $C = \frac{5!}{2!2!1!} = \frac{120}{2 \cdot 2 \cdot 1} = \frac{120}{4} = \mathbf{30}$

    \item \textbf{Coeficiente de $a^3 b^2 c$ em $(a + b + c)^6$:} \\
    $n=6, k_1=3, k_2=2, k_3=1$. \\
    $C = \frac{6!}{3!2!1!} = \frac{720}{6 \cdot 2 \cdot 1} = \frac{720}{12} = \mathbf{60}$

    \item \textbf{Coeficiente de $x^2 y z^2 w$ em $(x + y + z + w)^6$:} \\
    $n=6, k_1=2, k_2=1, k_3=2, k_4=1$. \\
    $C = \frac{6!}{2!1!2!1!} = \frac{720}{2 \cdot 1 \cdot 2 \cdot 1} = \frac{720}{4} = \mathbf{180}$
\end{enumerate}
\end{solucao}

% -------------------------------------------------------
\item \textbf{(Multinômio de Newton — Expansão Parcial)} Expanda completamente $(x + y + z)^3$:

\begin{solucao}
Listagem dos monômios e seus coeficientes ($k_1+k_2+k_3=3$):
\begin{itemize}
    \item $(3,0,0) \to x^3$. Coef: $\frac{3!}{3!0!0!} = 1$
    \item $(0,3,0) \to y^3$. Coef: $\frac{3!}{0!3!0!} = 1$
    \item $(0,0,3) \to z^3$. Coef: $\frac{3!}{0!0!3!} = 1$
    \item $(2,1,0) \to x^2y$. Coef: $\frac{3!}{2!1!0!} = 3$
    \item $(2,0,1) \to x^2z$. Coef: $\frac{3!}{2!0!1!} = 3$
    \item $(1,2,0) \to xy^2$. Coef: $\frac{3!}{1!2!0!} = 3$
    \item $(0,2,1) \to y^2z$. Coef: $\frac{3!}{0!2!1!} = 3$
    \item $(1,0,2) \to xz^2$. Coef: $\frac{3!}{1!0!2!} = 3$
    \item $(0,1,2) \to yz^2$. Coef: $\frac{3!}{0!1!2!} = 3$
    \item $(1,1,1) \to xyz$. Coef: $\frac{3!}{1!1!1!} = 6$
\end{itemize}
A expressão completa é: \\
$\mathbf{x^3 + y^3 + z^3 + 3x^2y + 3x^2z + 3xy^2 + 3y^2z + 3xz^2 + 3yz^2 + 6xyz}$
\end{solucao}

\end{enumerate}
