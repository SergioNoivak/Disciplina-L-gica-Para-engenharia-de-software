    \section{Introdução aos Grafos Completos}

    \begin{frame}{1. O que é um Grafo? (Revisão)}
        \begin{itemize}
            \item Um \textbf{grafo} $G = (V, E)$ consiste em um conjunto de vértices ($V$) e um conjunto de arestas ($E$).
            \item Vértices representam entidades (nós, pontos).
            \item Arestas representam relações ou conexões entre essas entidades.
            \item Na \textit{Lógica para Software}, grafos são estruturas fundamentais para modelar redes, dependências e fluxo de controle.
        \end{itemize}
        \vspace{0.5cm}
        \begin{center}
            \begin{tikzpicture}[scale=0.8, every node/.style={circle, draw, fill=blue!10, minimum size=0.6cm}]
                \node (A) at (0, 0) {$v_1$};

                \node (D) at (4, 0) {$v_4$};
                \draw (A) -- (B);
                \draw (A) -- (C);
                \draw (B) -- (C);
                \draw (B) -- (D);
            \end{tikzpicture}
        \end{center}
    \end{frame}

    \begin{frame}{2. Definição de Grafo Completo}
        \begin{block}{Definição}
            Um \textbf{grafo completo} é um grafo simples (sem laços ou arestas múltiplas) no qual \textbf{todo par de vértices distintos é conectado por uma única aresta}.
        \end{block}
        \vspace{0.5cm}
        \begin{itemize}
            \item Isso significa que existe uma ligação direta entre todos os elementos do conjunto.
            \item É o grafo ``mais denso'' possível para um determinado número de vértices.
            \item Representa um cenário onde ``todo mundo conhece todo mundo'' ou ``todos os nós estão interligados''.
        \end{itemize}
    \end{frame}

    \begin{frame}{3. Notação}
        \begin{itemize}
            \item Um grafo completo com $n$ vértices é denotado pela letra maiúscula \textbf{$K_n$}.
            \item O ``K'' é frequentemente associado ao termo alemão \textit{komplett} ou em homenagem ao matemático Kazimierz Kuratowski.
            \item Para especificar um grafo completo, basta saber o número de vértices $n$.
        \end{itemize}
        \vspace{0.5cm}
        \begin{alertblock}{Importante}
            Um subgrafo de qualquer grafo que por si só forma um grafo completo é chamado de \textbf{clique}.
        \end{alertblock}
    \end{frame}

    \begin{frame}{6. Propriedades: Número de Arestas}
        \begin{itemize}
            \item Como cada vértice se conecta a todos os outros vértices, o número de arestas cresce rapidamente com o número de vértices.
            \item O número de arestas em um grafo completo $K_n$ é dado pela combinação de $n$ elementos tomados 2 a 2:
        \end{itemize}
        \begin{block}{Fórmula do Número de Arestas}
            \[ |E| = \binom{n}{2} = \frac{n(n-1)}{2} \]
        \end{block}
        \begin{itemize}
            \item Exemplo para $K_5$: $n=5 \Rightarrow \frac{5 \cdot 4}{2} = 10$ arestas.
            \item Exemplo para $K_{10}$: $n=10 \Rightarrow \frac{10 \cdot 9}{2} = 45$ arestas.
        \end{itemize}
    \end{frame}

    \begin{frame}{7. Propriedades: Grau dos Vértices}
        \begin{itemize}
            \item O \textbf{grau} de um vértice é o número de arestas conectadas a ele.
            \item Em um grafo completo $K_n$, todos os vértices são simétricos e se conectam a todos os outros $n-1$ vértices do grafo.
        \end{itemize}
        \begin{block}{Grau em $K_n$}
            Em um grafo completo $K_n$, o grau de todo vértice é exatamente \textbf{$n - 1$}.
        \end{block}
        \begin{itemize}
            \item Como todos os vértices têm o mesmo grau, dizemos que $K_n$ é um \textbf{grafo regular} de grau $n-1$ (ou simplesmente $(n-1)$-regular).
        \end{itemize}
    \end{frame}

    \begin{frame}{8. Representação Computacional (Matriz de Adjacência)}
        \begin{itemize}
            \item Em computação, grafos são frequentemente representados por Matrizes de Adjacência.
            \item Para um grafo completo $K_n$, a matriz de adjacência $A_{n \times n}$ possui zeros na diagonal principal e uns em todas as outras posições.
        \end{itemize}
        \vspace{0.3cm}
        \textbf{Exemplo para $K_4$:}
        \[
        A_{K_4} = \begin{pmatrix}
        0 & 1 & 1 & 1 \\
        1 & 0 & 1 & 1 \\
        1 & 1 & 0 & 1 \\
        1 & 1 & 1 & 0 
        \end{pmatrix}
        \]
        \begin{itemize}
            \item Isso reflete que todo nó tem aresta com todos os outros, exceto com ele mesmo.
        \end{itemize}
    \end{frame}

    \begin{frame}{9. Aplicações em Redes e Software}
        \begin{itemize}
            \item \textbf{Redes de Computadores (Topologia de Malha Total):} Representa uma rede onde cada computador está conectado diretamente a todos os outros. Alta tolerância a falhas, mas custo altíssimo.
            \item \textbf{Problema do Caixeiro Viajante:} Encontrar o menor caminho passando por todos os vértices em um grafo completo com pesos nas arestas.
            \item \textbf{Sistemas Multiprocessados:} Conexões entre processadores e memória.
            \item \textbf{Redes Sociais (Cliques):} Modelagem de subgrupos onde todos os membros interagem entre si.
        \end{itemize}
    \end{frame}

    \section{Teoremas, Propriedades e Corolários}

    \begin{frame}{10. Introdução aos Teoremas em Grafos Completos}
        \begin{itemize}
            \item Grafos completos ($K_n$) são estruturas extremas na teoria dos grafos.
            \item Devido à sua densidade máxima de arestas, eles servem como casos base ou limites superiores para muitos teoremas.
            \item A seguir, exploraremos as principais propriedades estruturais, métricas de conectividade e comportamentos de travessia (Euler e Hamilton).
            \item Usaremos as definições de \textbf{Teorema} (proposição provada verdadeira) e \textbf{Corolário} (consequência direta de um teorema).
        \end{itemize}
    \end{frame}

    \begin{frame}{11. Teorema 1: O Lema do Aperto de Mãos}
        \begin{teorema}[Lema do Aperto de Mãos]
            Em qualquer grafo não direcionado $G = (V,E)$, a soma dos graus de todos os vértices é igual ao dobro do número de arestas:
            \[ \sum_{v \in V} \text{grau}(v) = 2|E| \]
        \end{teorema}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.6\textwidth}
                \begin{itemize}
                    \item \textbf{Explicação:} Cada aresta conecta exatamente dois vértices. Portanto, ao somar os graus de todos os vértices, cada aresta é contada exatamente duas vezes (uma para cada extremidade).
                    \item Este teorema é universal, mas tem uma aplicação muito direta nos grafos completos.
                \end{itemize}
            \end{column}
            \begin{column}{0.4\textwidth}
                \centering
                \begin{tikzpicture}[scale=0.8, every node/.style={circle, draw, fill=blue!10, minimum size=0.5cm, inner sep=1pt}]
                    \node (1) at (0,1) {1};
                    \node (2) at (1.5,1.5) {2};
                    \node (3) at (2,0) {3};
                    \node (4) at (0.5,-0.5) {4};
                    \draw (1) -- node[above, font=\tiny, red] {+2} (2);
                    \draw (2) -- node[right, font=\tiny, red] {+2} (3);
                    \draw (1) -- node[left, font=\tiny, red] {+2} (4);
                \end{tikzpicture}\\
                \tiny $\Sigma$ Graus = $2+2+1+1 = 6$\\$2 \times |E| = 2 \times 3 = 6$
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{12. Corolário 1: Número de Arestas em $K_n$}
        \begin{block}{Corolário}
            Um grafo completo $K_n$ possui exatamente $\frac{n(n-1)}{2}$ arestas.
        \end{block}
        \vspace{0.3cm}
        \begin{itemize}
            \item Como vimos anteriormente, essa propriedade deriva diretamente do Lema do Aperto de Mãos e do fato de $K_n$ ser regular.
        \end{itemize}
    \end{frame}

    \begin{frame}{13. Prova do Corolário 1}
        \begin{proof}
            \begin{enumerate}
                \item Seja $K_n$ um grafo completo com $n$ vértices.
                \item Por definição, cada vértice em $K_n$ é conectado a todos os outros $n-1$ vértices. Assim, $\text{grau}(v) = n-1$ para todo $v \in V$.
                \item Aplicando o Lema do Aperto de Mãos:
                \[ \sum_{v \in V} \text{grau}(v) = 2|E| \]
                \item Como existem $n$ vértices, a soma é $n \times (n-1)$. Logo:
                \[ n(n-1) = 2|E| \implies |E| = \frac{n(n-1)}{2} \]
            \end{enumerate}
        \end{proof}
    \end{frame}

    \begin{frame}{19. Propriedade 3: Ciclos Hamiltonianos}
        \begin{itemize}
            \item \textbf{Ciclo Hamiltoniano:} Um ciclo que visita \textbf{cada vértice} de um grafo exatamente uma vez, voltando ao início.
            \item O Problema do Caixeiro Viajante busca o ciclo hamiltoniano de menor custo.
            \item Diferente do caminho Euleriano, encontrar um ciclo Hamiltoniano em um grafo genérico é um problema NP-Completo.
            \item \textbf{Mas e em grafos completos?}
        \end{itemize}
    \end{frame}

    \begin{frame}{21. Corolário 3: Número de Ciclos Hamiltonianos em $K_n$}
        \begin{block}{Corolário}
            O número de ciclos Hamiltonianos distintos em um grafo completo $K_n$ é $\frac{(n-1)!}{2}$.
        \end{block}
        \vspace{0.3cm}
        \begin{proof}
            \begin{enumerate}
                \item Existem $n!$ permutações possíveis para visitar os $n$ vértices.
                \item Como é um ciclo, o vértice inicial não importa (uma rota ABC é o mesmo ciclo que BCA e CAB), então dividimos por $n$, resultando em $(n-1)!$.
                \item Como o grafo não é direcionado, percorrer o ciclo num sentido (A-B-C) ou no oposto (C-B-A) gera o mesmo conjunto de arestas, então dividimos por $2$. Resulta em $\frac{(n-1)!}{2}$.
            \end{enumerate}
        \end{proof}
    \end{frame}


    \begin{frame}{29. Teorema 8: Árvores Geradoras em $K_n$}
        \begin{itemize}
            \item Uma \textbf{árvore geradora} de um grafo é um subgrafo que é uma árvore (sem ciclos) e que conecta todos os vértices do grafo original.
        \end{itemize}
        \begin{teorema}[Fórmula de Cayley]
            O número de árvores geradoras distintas em um grafo completo rotulado $K_n$ é dado por: $n^{n-2}$

        \vspace{0.2cm}
        \begin{itemize}
            \item Exemplo para $K_3$ (triângulo): O número de árvores geradoras é $3^{3-2} = 3^1 = 3$. (Basta remover uma das 3 arestas do triângulo).

    \begin{frame}{31. Introdução aos Grafos Bipartidos}

    \begin{frame}{32. Definição Formal}
        \begin{block}{Grafo Bipartido}
            Um grafo $G = (V, E)$ é bipartido se o seu conjunto de vértices $V$ pode ser particionado em dois subconjuntos disjuntos $U$ e $W$ (ou seja, $U \cup W = V$ e $U \cap W = \emptyset$), de tal forma que toda aresta em $E$ conecte um vértice em $U$ a um vértice em $W$.
        \end{block}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.5\textwidth}
                \begin{itemize}
                    \item \textbf{Regra de Ouro:} Nenhuma aresta conecta dois vértices do mesmo conjunto.
                    \item $U$ e $W$ são frequentemente chamados de ``lados'' ou ``partições'' do grafo.
                \end{itemize}
            \end{column}
            \begin{column}{0.5\textwidth}
                \centering
                \begin{tikzpicture}
                    \draw[fill=green!10, rounded corners] (-0.5,-0.5) rectangle (0.5, 2.5);
                    \draw[fill=orange!10, rounded corners] (1.5,-0.5) rectangle (2.5, 2.5);
                    \node at (0, 2.8) {$U$};
                    \node at (2, 2.8) {$W$};
                    
                    \node[circle, draw, fill=green!40, minimum size=0.4cm, inner sep=1pt] (u1) at (0,2) {};
                    \node[circle, draw, fill=green!40, minimum size=0.4cm, inner sep=1pt] (u2) at (0,1) {};
                    \node[circle, draw, fill=green!40, minimum size=0.4cm, inner sep=1pt] (u3) at (0,0) {};
                    
                    \node[circle, draw, fill=orange!40, minimum size=0.4cm, inner sep=1pt] (w1) at (2,1.5) {};
                    \node[circle, draw, fill=orange!40, minimum size=0.4cm, inner sep=1pt] (w2) at (2,0.5) {};
                    
                    \draw (u1)--(w1) (u2)--(w1) (u2)--(w2) (u3)--(w2);
                    \draw[red, thick, dashed] (u1) to[bend right] node[left] {\tiny Proibido!} (u2);
                \end{tikzpicture}
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{34. Propriedade Fundamental: 2-Colorabilidade}
        \begin{teorema}
            Um grafo é bipartido se e somente se o seu Número Cromático $\chi(G)$ for menor ou igual a 2.
        \end{teorema}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.6\textwidth}
                \begin{itemize}
                    \item Isso significa que você sempre consegue pintar os vértices de um grafo bipartido usando apenas \textbf{duas cores}, sem que dois vértices da mesma cor fiquem conectados.
                    \item Na prática: Pinte todos os vértices de $U$ com a Cor 1 e todos de $W$ com a Cor 2. Como não há arestas dentro de $U$ nem dentro de $W$, a regra de coloração é respeitada perfeitamente.
                \end{itemize}
            \end{column}
            \begin{column}{0.4\textwidth}
                \centering
                \begin{tikzpicture}[every node/.style={circle, draw, minimum size=0.5cm, inner sep=1pt}]
                    \node[fill=blue!30] (A) at (0,1.5) {};
                    \node[fill=red!30] (B) at (1.5,1.5) {};
                    \node[fill=blue!30] (C) at (1.5,0) {};
                    \node[fill=red!30] (D) at (0,0) {};
                    \draw (A)--(B)--(C)--(D)--(A);
                \end{tikzpicture}\\
                \vspace{0.2cm}
                \tiny $\chi(G) = 2$ (Apenas Azul e Vermelho)
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{35. Condição Necessária: Ausência de Ciclos Ímpares}
        \begin{teorema}[Teorema de König]
            Um grafo é bipartido se e somente se ele \textbf{não possui nenhum ciclo de comprimento ímpar} (ciclos com 3, 5, 7... arestas).
        \end{teorema}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.6\textwidth}
                \begin{itemize}
                    \item Se houvesse um ciclo ímpar (ex: um triângulo), ao tentar pintar o grafo com 2 cores, você inevitavelmente chegaria em dois vértices adjacentes que precisariam ter a mesma cor.
                    \item Assim que você encontra um ciclo ímpar (ex: subgrafo $K_3$) em qualquer rede, você tem certeza matemática de que ela \textbf{não} é bipartida.
                \end{itemize}
            \end{column}
            \begin{column}{0.4\textwidth}
                \centering
                \begin{tikzpicture}[every node/.style={circle, draw, minimum size=0.4cm, inner sep=1pt}]
                    \node[fill=blue!30] (1) at (90:1cm) {};

                \tiny Ciclo de 5 arestas (Falha ao colorir)
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{36. O Grafo Bipartido Completo ($K_{m,n}$)}
        \begin{block}{Definição}
            Um \textbf{Grafo Bipartido Completo}, denotado por $K_{m,n}$, é um grafo bipartido onde \textbf{cada} vértice do conjunto $U$ está conectado a \textbf{todos} os vértices do conjunto $W$.
        \end{block}
        \vspace{0.3cm}
        \begin{itemize}
            \item Sendo $m = |U|$ (número de vértices em $U$) e $n = |W|$ (número de vértices em $W$).
            \item É a estrutura ``mais densa'' possível sem violar a regra de ser bipartido.
            \item Faz paralelo com o $K_n$, mas focado nas relações extremas entre dois grupos.
        \end{itemize}
    \end{frame}

    \begin{frame}{38. Propriedades de $K_{m,n}$}
        \begin{itemize}
            \item \textbf{Número de Vértices:} O total de vértices em $K_{m,n}$ é simplesmente a soma $m + n$.
            \item \textbf{Número de Arestas:} Como cada vértice dos $m$ se conecta a cada um dos $n$, o número total de arestas é dado pelo produto cartesiano das partições:
        \end{itemize}
        \begin{columns}
            \begin{column}{0.6\textwidth}
                \begin{block}{Fórmula do Número de Arestas em $K_{m,n}$}
                    \[ |E| = m \times n \]
                \end{block}
                \begin{itemize}
                    \item Exemplo: $K_{3,3}$ possui $3+3 = 6$ vértices e $3 \times 3 = 9$ arestas.
                    \item Todos os vértices do conjunto de tamanho $m$ possuem grau $n$, e vice-versa.
                \end{itemize}
            \end{column}
            \begin{column}{0.4\textwidth}
                \centering

    \section{Estruturas Hierárquicas: Árvores Binárias}

    \begin{frame}{41. O que é uma Árvore (em Grafos)?}
        \begin{columns}
            \begin{column}{0.6\textwidth}
                \begin{itemize}
                    \item Na teoria dos grafos, uma \textbf{Árvore} é um grafo conexo e \textbf{acíclico} (não possui ciclos).
                    \item Se você adicionar apenas \textbf{uma aresta} a uma árvore, você cria um ciclo. Se remover uma aresta, você desconecta o grafo.
                    \item Em Engenharia de Software, as árvores são usadas para modelar \textbf{hierarquia} (ex: sistemas de pastas do Windows/Linux, hierarquias de classes na Orientação a Objetos).
                \end{itemize}
            \end{column}
            \begin{column}{0.4\textwidth}

    \begin{frame}{42. A Árvore Binária}
        \begin{block}{Definição}
            Uma \textbf{Árvore Binária} é uma árvore com raiz na qual cada nó tem \textbf{no máximo dois filhos}, tipicamente chamados de \textit{filho esquerdo} e \textit{filho direito}.
        \end{block}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.5\textwidth}
                \begin{itemize}
                    \item Diferente dos grafos completos ou bipartidos, onde os nós têm muitos vizinhos ``iguais'', na árvore binária existe uma relação rígida de \textbf{pai-filho}.
                    \item Isso a torna extremamente eficiente para decisões de ``sim/não'' ou ``maior/menor''.
                \end{itemize}
            \end{column}
            \begin{column}{0.5\textwidth}
                \centering
                \begin{tikzpicture}[level distance=1cm,
                    level 1/.style={sibling distance=1.5cm},
                    every node/.style={circle, draw, fill=blue!10, minimum size=0.4cm, inner sep=1pt, font=\tiny}]
                    \node {Pai}
                    child {node {Esq.}}
                    child {node {Dir.}};
                \end{tikzpicture}
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{43. Anatomia de uma Árvore Binária}
        \begin{columns}
            \begin{column}{0.5\textwidth}
                \begin{itemize}
                    \item \textbf{Raiz (Root):} O topo da árvore (nível 0).
                    \item \textbf{Nós Internos:} Possuem pelo menos 1 filho.
                    \item \textbf{Folhas (Leaves):} Nós sem filhos (o fim da linha).
                    \item \textbf{Altura (Height):} A maior distância da raiz até a folha mais profunda.
                \end{itemize}
            \end{column}

    \begin{frame}{44. Árvore Binária de Busca (BST)}
        \begin{columns}
            \begin{column}{0.6\textwidth}
                \begin{itemize}
                    \item Uma das aplicações mais geniais das árvores binárias é a \textbf{Binary Search Tree (BST)}.
                    \item Ela impõe uma \textbf{regra de ordenação lógica}:
                \end{itemize}
                \begin{block}{A Regra de Ouro da BST}
                    Para todo nó $N$: \\
                    1. Todos os valores à \textbf{esquerda} de $N$ são \textbf{menores} que o valor de $N$. \\
                    2. Todos os valores à \textbf{direita} de $N$ são \textbf{maiores} que o valor de $N$.
                \end{block}
            \end{column}
            \begin{column}{0.4\textwidth}
                \centering
                \begin{tikzpicture}[level distance=1.2cm,
                    level 1/.style={sibling distance=1.5cm},
                    every node/.style={circle, draw, fill=orange!20, minimum size=0.6cm, inner sep=1pt}]
                    \node {$N$}
                    child {node {$< N$}}
                    child {node {$> N$}};
                \end{tikzpicture}
            \end{column}
        \end{columns}
    \end{frame}

    \section{Teoremas e Propriedades das Árvores Binárias}

    \begin{frame}{46. Classificação: Árvore Binária Estrita (Full)}
        \begin{block}{Definição}
            Uma \textbf{Árvore Binária Estrita} (ou Cheia/Própria) é aquela em que \textbf{todo nó tem exatamente 0 ou 2 filhos}. Nenhum nó interno tem apenas 1 filho.
        \end{block}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.5\textwidth}
                \begin{itemize}
                    \item Utilizada frequentemente para representar expressões matemáticas rigorosas (onde todo operador binário exige 2 operandos).

    \begin{frame}{47. Classificação: Árvore Binária Completa}

    \begin{frame}{48. Classificação: Árvore Binária Perfeita}
        \begin{block}{Definição}
            Uma \textbf{Árvore Binária Perfeita} é aquela em que \textbf{todos os nós internos têm 2 filhos} e \textbf{todas as folhas estão exatamente no mesmo nível}.
        \end{block}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.5\textwidth}
                \begin{itemize}
                    \item É a árvore binária mais densa possível para uma dada altura.
                    \item Extremamente rara na prática, mas fundamental para a análise de complexidade (o melhor caso).
                \end{itemize}

    \begin{frame}{49. Teorema 1: Nós Máximos por Nível}
        \begin{teorema}
            O número máximo de nós em um determinado nível $k$ de uma árvore binária é \textbf{$2^k$}, considerando a raiz no nível $k=0$.
        \end{teorema}
        \vspace{0.3cm}
        \begin{itemize}
            \item \textbf{Nível 0 (Raiz):} $2^0 = 1$ nó.
            \item \textbf{Nível 1:} $2^1 = 2$ nós (filhos da raiz).
            \item \textbf{Nível 2:} $2^2 = 4$ nós.
            \item \textbf{Nível 3:} $2^3 = 8$ nós.
            \item Como a ramificação é binária (base 2), a cada nível a capacidade dobra. O crescimento é exponencial.
        \end{itemize}
    \end{frame}

    \begin{frame}{50. Teorema 2: Nós em uma Árvore Perfeita}
        \begin{teorema}
            O número total de nós ($N$) de uma árvore binária perfeita de altura $h$ é a soma de uma progressão geométrica: \textbf{$N = 2^{h+1} - 1$}.
        \end{teorema}
        \vspace{0.3cm}
        \begin{itemize}
            \item Para uma árvore perfeita com altura $h=2$ (Raiz + 2 níveis abaixo):
            \item $N = 2^{2+1} - 1 = 2^3 - 1 = 8 - 1 = \textbf{7}$ nós.
            \item Isso prova matematicamente o porquê de buscas binárias serem tão rápidas: com apenas altura 20 ($h=20$), podemos armazenar $2^{21} - 1 \approx 2$ \textbf{milhões} de nós!
        \end{itemize}
    \end{frame}

    \begin{frame}{51. Teorema 3: Árvore Degenerada (Pior Caso)}
        \begin{teorema}
            O número mínimo de nós ($N$) em uma árvore binária de altura $h$ é \textbf{$N = h + 1$}.
        \end{teorema}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.6\textwidth}
                \begin{itemize}
                    \item Isso acontece quando cada nó tem \textbf{apenas 1 filho}.
                    \item A árvore binária sofre uma "degeneração" e se transforma em uma simples \textbf{Lista Encadeada} (\textit{Linked List}).
                    \item Toda a vantagem do $O(\log N)$ é perdida, voltando a ser uma busca ineficiente $O(N)$.
                \end{itemize}
            \end{column}
            \begin{column}{0.4\textwidth}
                \centering
                \begin{tikzpicture}[level distance=0.8cm,
                    level 1/.style={sibling distance=1cm},
                    every node/.style={circle, draw, fill=red!20, minimum size=0.4cm, inner sep=1pt}]

                    child[missing] {}
                    child {node {2}
                        child[missing] {}
                        child {node {3}
                        child[missing] {}
                        child {node {4}}
                        }
                    };
                \end{tikzpicture}\\
                \vspace{0.2cm}
                \tiny Árvore Degenerada (Pior caso)
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{52. Propriedade: Folhas vs. Nós Internos}
        \begin{teorema}
            Em qualquer \textbf{Árvore Binária Estrita} (onde nós têm 0 ou 2 filhos), o número de folhas ($L$) é sempre igual ao número de nós internos ($I$) mais um: \textbf{$L = I + 1$}.
        \end{teorema}
        \vspace{0.3cm}
        \begin{columns}
            \begin{column}{0.6\textwidth}
                \begin{itemize}
                    \item Não importa a estrutura da árvore. Se todo nó ramifica para 2, a base será exatamente 1 unidade maior que os nós do meio.
                    \item Exemplo prático: Torneios de eliminação (chaveamentos esportivos). Para haver 8 times na base (Folhas), são necessários 7 jogos (Nós Internos).
                \end{itemize}
            \end{column}
            \begin{column}{0.4\textwidth}
                \centering
                \begin{tikzpicture}[level distance=0.8cm,
                    level 1/.style={sibling distance=1.5cm},
                    level 2/.style={sibling distance=0.8cm},

                        child {node[fill=green!20] {L}}
                        child {node[fill=green!20] {L}}
                    }
                    child {node[fill=gray!20] {I}
                        child {node[fill=green!20] {L}}
                        child {node[fill=green!20] {L}}
                    };
                \end{tikzpicture}\\
                \vspace{0.2cm}
                \tiny $I=3$ e $L=4$
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{53. Teorema das Arestas em Árvores}
        \begin{teorema}
            Toda árvore (binária ou genérica) com $N$ vértices possui exatamente \textbf{$N - 1$ arestas}.
        \end{teorema}
        \vspace{0.3cm}
        \begin{itemize}
            \item Consequência direta do Lema do Aperto de Mãos. 
            \item Como a árvore é conectada e acíclica, para conectar o primeiro nó a um novo nó, você precisa de exatamente 1 aresta extra.
            \item Se houvesse menos que $N-1$ arestas, o grafo estaria \textbf{desconectado}.
            \item Se houvesse $N$ ou mais arestas, inevitavelmente se formaria um \textbf{ciclo}.
        \end{itemize}
    \end{frame}

    \begin{frame}{54. A Relação Logarítmica: $h \approx \log_2(N)$}
        \begin{block}{Equação da Altura Mínima}
            A altura mínima $h_{min}$ para acomodar $N$ nós é: \textbf{$h \approx \lceil \log_2(N+1) \rceil - 1$}
        \end{block}
        \vspace{0.3cm}
        \begin{itemize}
            \item Na computação, a altura $h$ de uma árvore representa o \textbf{número máximo de "ifs" ou "saltos"} necessários para encontrar uma informação.
            \item O crescimento logarítmico significa que o volume de dados pode crescer brutalmente, mas a altura da árvore cresce de forma ínfima.
            \item Essa é a prova matemática do porquê bancos de dados e sistemas de busca são viáveis em larga escala.
        \end{itemize}
    \end{frame}

    \begin{frame}{55. Conclusão: O Desafio do Balanceamento}
        \begin{itemize}
            \item Estudamos os \textbf{Teoremas} e vimos os \textbf{Extremos}:
            \item \textbf{Melhor Caso (Árvore Perfeita):} A busca é $O(\log N)$.
            \item \textbf{Pior Caso (Árvore Degenerada):} A busca é $O(N)$.
        \end{itemize}
        \vspace{0.2cm}
        \begin{block}{O que os Cientistas da Computação Fizeram?}
            Inventaram Árvores de Busca \textbf{Auto-Balanceáveis} (Árvore AVL, Red-Black). Toda vez que um nó é inserido, algoritmos aplicam "Rotações" na árvore para garantir matematicamente que ela se mantenha o mais próxima possível de uma Árvore Perfeita, salvaguardando a performance $O(\log N)$.
        \end{block}
    \end{frame}
