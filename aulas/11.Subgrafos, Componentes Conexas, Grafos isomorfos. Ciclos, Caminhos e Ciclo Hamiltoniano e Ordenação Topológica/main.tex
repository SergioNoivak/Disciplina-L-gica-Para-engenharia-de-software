\section{Subgrafos}
\subsection{Conceitos Iniciais}

\begin{frame}{Subgrafos: Definição Formal}
    \begin{block}{Definição}
        Seja $G = (V, E)$ um grafo. Um grafo $H = (V_H, E_H)$ é dito um \textbf{subgrafo} de $G$, denotado por $H \subseteq G$, se e somente se:
        \begin{itemize}
            \item $V_H \subseteq V$
            \item $E_H \subseteq E$
            \item $\forall e \in E_H$, se $e = \{u, v\}$, então $\{u, v\} \subseteq V_H$
        \end{itemize}
    \end{block}
    \vspace{0.3cm}
    A terceira condição (fecho das arestas) garante matematicamente que toda aresta em $H$ incide exclusivamente sobre vértices que também estão presentes em $H$, mantendo a estrutura de grafo bem definida.
\end{frame}

\begin{frame}{Exemplo Prático: Grafo Original $G$}
    Considere o seguinte grafo $G = (V, E)$:
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, node distance=2cm, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (A) at (0,2) {A};
        \node (B) at (2,2) {B};
        \node (C) at (0,0) {C};
        \node (D) at (2,0) {D};
        
        \path[thick] 
        (A) edge (B)
        (A) edge (C)
        (B) edge (C)
        (B) edge (D)
        (C) edge (D);
    \end{tikzpicture}
    \end{center}
    $V = \{A, B, C, D\}$ \\
    $E = \{\{A,B\}, \{A,C\}, \{B,C\}, \{B,D\}, \{C,D\}\}$
\end{frame}

\begin{frame}{Exemplo Prático: Subgrafo $H$}
    A partir de $G$, podemos formar o subgrafo $H = (V_H, E_H)$:
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, node distance=2cm, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (A) at (0,2) {A};
        \node (B) at (2,2) {B};
        \node (C) at (0,0) {C};
        
        \path[thick] 
        (A) edge (B)
        (A) edge (C);
    \end{tikzpicture}
    \end{center}
    $V_H = \{A, B, C\} \subseteq \{A, B, C, D\}$ \\
    $E_H = \{\{A,B\}, \{A,C\}\} \subseteq E$
\end{frame}

\begin{frame}{Tipos Especiais de Subgrafos}
    \begin{block}{Subgrafo Próprio}
        Um subgrafo $H$ de $G$ é chamado de \textbf{subgrafo próprio} se $H \neq G$. Ou seja, $H$ tem menos vértices ou menos arestas que $G$.
    \end{block}
    
    \begin{block}{Subgrafo Gerador (Spanning Subgraph)}
        Um subgrafo $H$ é um \textbf{subgrafo gerador} de $G$ se ele contém \textbf{todos os vértices} de $G$ ($V_H = V$), mas pode ter menos arestas ($E_H \subseteq E$).
    \end{block}
\end{frame}

\begin{frame}{Exemplo: Subgrafo Gerador}
    Subgrafo $H'$ onde $V_{H'} = \{A, B, C, D\}$ e $E_{H'} = \{\{A,B\}, \{A,C\}, \{B,D\}, \{C,D\}\}$.
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (A) at (0,2) {A};
        \node (B) at (2,2) {B};
        \node (C) at (0,0) {C};
        \node (D) at (2,0) {D};
        
        \path[thick] 
        (A) edge (B)
        (A) edge (C)
        (B) edge (D)
        (C) edge (D);
    \end{tikzpicture}
    \end{center}
    Aqui, removemos apenas a aresta $\{B,C\}$. Como todos os vértices originais estão presentes, este é um \textbf{subgrafo gerador}.
\end{frame}

\begin{frame}{Subgrafo Induzido}
    \begin{block}{Definição}
        Um subgrafo $H$ de $G$ é dito \textbf{induzido} por um conjunto de vértices $V_H \subseteq V$ se $H$ contém \textbf{todas} as arestas de $G$ que ligam vértices de $V_H$.
    \end{block}
    \vspace{0.5cm}
    \textbf{Exemplo}: Subgrafo induzido pelo conjunto $V_H = \{A, B, C\}$. \\
    Em $G$, as arestas entre $\{A, B, C\}$ são $\{A,B\}, \{A,C\}$ e $\{B,C\}$.
    \begin{center}
    \begin{tikzpicture}[scale=1, every node/.style={circle, draw, minimum size=6mm, thick}]
        \node (A) at (0,2) {A};
        \node (B) at (2,2) {B};
        \node (C) at (0,0) {C};
        
        \path[thick] 
        (A) edge (B)
        (A) edge (C)
        (B) edge (C);
    \end{tikzpicture}
    \end{center}
    (O subgrafo $H$ dos primeiros slides \textbf{não} era induzido, pois faltava a aresta $\{B,C\}$).
\end{frame}

\begin{frame}{Resumo e Aplicações}
    \begin{itemize}
        \item \textbf{Subgrafo}: Qualquer subconjunto válido de vértices e arestas.
        \item \textbf{Subgrafo Gerador}: Mantém todos os vértices, podendo omitir arestas (muito usado em árvores geradoras, roteamento).
        \item \textbf{Subgrafo Induzido}: Escolhe-se os vértices e obrigatoriamente mantém-se todas as arestas originais entre eles (útil para encontrar cliques, comunidades em redes).
    \end{itemize}
    \vspace{0.5cm}
    Compreender subgrafos é o primeiro passo para analisar estruturas menores dentro de grandes redes complexas.
\end{frame}

\section{Componentes Conexas}
\subsection{Definição e Identificação}
\begin{frame}{Componentes Conexas: Definição}
    \begin{block}{Definição}
        Uma \textbf{componente conexa} de um grafo é um conjunto \textbf{maximal} de vértices que estão conectados por caminhos.
    \end{block}
    \vspace{0.5cm}
    \textbf{O que significa "maximal"?} \\
    Significa que não é possível adicionar mais nenhum outro vértice do grafo a esse conjunto sem perder a propriedade de que todos estão conectados entre si por algum caminho.
    \vspace{0.3cm}
    \begin{itemize}
        \item Se o grafo todo é conexo, ele possui apenas \textbf{1} componente conexa.
        \item Se o grafo é desconexo, ele possui \textbf{2 ou mais} componentes conexas.
    \end{itemize}
\end{frame}

\begin{frame}{Exemplo Visual: Grafo Desconexo}
    Considere o seguinte grafo $G$:
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.2, node distance=2cm, every node/.style={circle, draw, minimum size=8mm, thick}]
        % Componente 1
        \node (A) at (0,2) {A};
        \node (B) at (2,2) {B};
        \node (C) at (4,2) {C};
        
        \path[thick] 
        (A) edge (B)
        (B) edge (C);
        
        % Componente 2
        \node (D) at (0,0) {D};
        \node (E) at (2,0) {E};

        (D) edge (E);

    \end{tikzpicture}

    \vspace{0.5cm}
    Embora pareçam três figuras separadas, matematicamente isso compõe \textbf{um único grafo} $G$.
\end{frame}

\begin{frame}{Aplicações Práticas}
    O estudo de componentes conexas é fundamental na Engenharia de Software e em outras áreas para entender o isolamento e a comunicação em sistemas:
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Redes Sociais}: Identificar bolhas ou comunidades isoladas (grupos de pessoas onde ninguém conhece ninguém fora do grupo).
        \vspace{0.3cm}
        \item \textbf{Redes Elétricas}: Detectar partes da malha elétrica que ficaram sem energia ou isoladas após a falha de uma conexão principal.
        \vspace{0.3cm}
        \item \textbf{Sistemas de Transporte}: Verificar se todas as cidades de uma malha viária ou aeroportuária são acessíveis a partir de qualquer outra.
    \end{itemize}
\end{frame}

\section{Grafos Isomorfos}
\subsection{A Intuição do Isomorfismo}
\begin{frame}{Grafos Isomorfos}
    \begin{block}{Objetivos}
        \begin{itemize}
            \item Compreender o conceito de \textbf{isomorfismo} em teoria dos grafos.
            \item Aplicar técnicas e estratégias de verificação para determinar se dois grafos são isomorfos.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{A Intuição do Isomorfismo}
    \begin{itemize}
        \item Muitas vezes, dois grafos desenhados de maneiras completamente diferentes na verdade representam a \textbf{mesma estrutura}.
        \item Se pudermos "mover" os vértices de um grafo (esticando ou dobrando as arestas, mas sem cortá-las) de modo que ele fique exatamente igual ao outro, os dois grafos são \textbf{isomorfos}.
        \item Isomorfismo vem do grego: \textit{isos} (igual) e \textit{morphos} (forma).
    \end{itemize}
\end{frame}

\begin{frame}{Definição}
    \begin{block}{Definição Formal}
        Dois grafos $G$ e $H$ são \textbf{isomorfos} quando possuem exatamente a \textbf{mesma estrutura} topológica, mesmo que seus vértices tenham nomes diferentes ou sejam desenhados de formas diferentes.
    \end{block}
    \vspace{0.5cm}
    Em outras palavras, se os grafos forem "despidos" dos rótulos de seus vértices, não haverá como distinguir um do outro.
\end{frame}

\begin{frame}{A Função Bijetora}
    Matematicamente, $G$ e $H$ são isomorfos se existir uma \textbf{função bijetora}:
    \vspace{0.3cm}
    \begin{center}
        \Large $f: V(G) \rightarrow V(H)$
    \end{center}
    \vspace{0.3cm}
    Que \textbf{preserva as adjacências}. Isso significa que:
    \begin{itemize}
        \item Dois vértices $u, v$ são vizinhos em $G$ \textbf{se, e somente se}, $f(u)$ e $f(v)$ são vizinhos em $H$.
        \item A função pareia cada vértice de $G$ com exatamente um vértice de $H$ e vice-versa.
    \end{itemize}
\end{frame}

\begin{frame}{Exemplo: Grafo $G$}
    Considere o grafo $G$:
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (A) at (0,2) {A};
        \node (B) at (2,2) {B};
        \node (C) at (2,0) {C};
        \node (D) at (0,0) {D};
        
        \path[thick] 
        (A) edge (B)
        (B) edge (C)
        (C) edge (D)
        (D) edge (A);
    \end{tikzpicture}
    \end{center}
    Este é um ciclo com 4 vértices (um quadrado).
\end{frame}

\begin{frame}{Exemplo: Grafo $H$}
    Agora considere o grafo $H$:
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (1) at (0,2) {1};
        \node (2) at (2,2) {2};
        \node (3) at (2,0) {3};
        \node (4) at (0,0) {4};
        
        \path[thick] 
        (1) edge (2)
        (2) edge (3)
        (3) edge (4)
        (4) edge (1);
    \end{tikzpicture}
    \end{center}
    À primeira vista, os rótulos são diferentes, mas a estrutura de conexões é idêntica ao grafo $G$.
\end{frame}

\begin{frame}{Mapeamento de Adjacências ($G \rightarrow H$)}
    Para provar o isomorfismo, aplicamos o seguinte mapeamento (a função bijetora $f$):
    \vspace{0.5cm}
    \begin{center}
        \begin{tabular}{c c c}
            \textbf{Grafo $G$} & $\rightarrow$ & \textbf{Grafo $H$} \\
            A & $\rightarrow$ & 1 \\
            B & $\rightarrow$ & 2 \\
            C & $\rightarrow$ & 3 \\
            D & $\rightarrow$ & 4 \\
        \end{tabular}
    \end{center}
    \vspace{0.5cm}
    Verifique as adjacências: A e B estão conectados, assim como 1 e 2 estão. C e D estão conectados, assim como 3 e 4 estão. Todas as conexões são perfeitamente preservadas.
\end{frame}

\begin{frame}{Estratégia de Verificação}
    Antes de tentar encontrar o mapeamento, podemos verificar rapidamente algumas \textbf{condições necessárias}.
    
    Dois grafos \textbf{só podem} ser isomorfos se tiverem:
    \begin{itemize}
        \item O mesmo \textbf{número de vértices} ($|V(G)| = |V(H)|$).
        \item O mesmo \textbf{número de arestas} ($|E(G)| = |E(H)|$).
        \item A mesma \textbf{sequência de graus} (os graus dos vértices ordenados devem ser idênticos).
        \item A mesma \textbf{estrutura de ciclos} (ex: mesmo número de triângulos, quadrados).
    \end{itemize}
    \vspace{0.3cm}
    \textit{Nota: Se alguma condição falhar, os grafos \textbf{não} são isomorfos.}
\end{frame}

\begin{frame}{Algoritmo Combinatorial Simples}
    Caso as condições iniciais sejam atendidas, o algoritmo básico (força bruta) para provar o isomorfismo é:
    \vspace{0.3cm}
    \begin{enumerate}
        \item Compare a quantidade de vértices.
        \item Compare a quantidade de arestas.
        \item Compare a sequência de graus.
        \item \textbf{Teste permutações:} Gere todas as formas possíveis de parear os vértices de $G$ com os vértices de $H$.
        \item \textbf{Verifique adjacências:} Para cada permutação, teste se as adjacências são preservadas. O primeiro mapeamento que funcionar confirma o isomorfismo.
    \end{enumerate}
\end{frame}

\begin{frame}{Complexidade do Algoritmo Combinatorial}
    O passo de testar permutações é o grande gargalo da verificação de isomorfismo em computação:
    \vspace{0.5cm}
    \begin{block}{Complexidade}
        A complexidade deste algoritmo força bruta é \textbf{$\mathcal{O}(n!)$} (onde $n$ é o número de vértices).
    \end{block}
    \vspace{0.5cm}
    \begin{itemize}
        \item Para um grafo com 10 vértices: $10! = 3.628.800$ permutações.
        \item Para um grafo com 20 vértices: $\approx 2.4 \times 10^{18}$ permutações (impraticável!).
    \end{itemize}
    \vspace{0.3cm}
    É por isso que o \textit{Problema do Isomorfismo de Grafos} é famoso na Ciência da Computação.
\end{frame}

\section{Caminhos}

        \begin{itemize}
            \item Diferenciar os conceitos de \textbf{caminhos} e \textbf{ciclos} em grafos.
            \item Identificar e compreender o que torna um ciclo \textbf{Hamiltoniano}.
            \item Compreender como esses conceitos são aplicados para resolver problemas clássicos da Computação.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{O que é um Caminho?}
    \begin{block}{Definição de Caminho}
        Um \textbf{caminho} em um grafo é uma sequência de vértices adjacentes conectados por arestas, onde nos movemos de um vértice para outro seguindo a estrutura do grafo.
    \end{block}
    \vspace{0.5cm}
    \begin{itemize}
        \item É literalmente a "rota" que pode ser percorrida partindo de um nó origem até um nó destino.
        \item Formalmente, é uma sequência $v_1, v_2, ..., v_k$ onde cada par adjacente $(v_i, v_{i+1})$ é uma aresta do grafo.
    \end{itemize}
\end{frame}

\begin{frame}{Exemplo Visual: Caminho}
    \textbf{Exemplo:} O caminho $A \rightarrow B \rightarrow C \rightarrow D$
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, node distance=2cm, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (A) at (0,0) {A};
        \node (B) at (2,0) {B};
        \node (C) at (4,0) {C};
        \node (D) at (6,0) {D};
        \node (E) at (3,1) {E};
        
        \path[thick] 
        (A) edge[->, line width=1.5pt, red] (B)
        (B) edge[->, line width=1.5pt, red] (C)
        (C) edge[->, line width=1.5pt, red] (D)
        (B) edge (E)
        (C) edge (E);
    \end{tikzpicture}
    \end{center}
    \vspace{0.3cm}
    As setas em \textbf{vermelho} destacam a sequência de vértices visitados.
\end{frame}

\begin{frame}{Comprimento de um Caminho}
    \begin{block}{Como medimos um caminho?}
        O \textbf{comprimento} de um caminho é determinado pelo \textbf{número de arestas} que o compõem, e não pelo número de vértices.
    \end{block}
    \vspace{0.5cm}
    Voltando ao exemplo $A \rightarrow B \rightarrow C \rightarrow D$:
    \begin{itemize}
        \item Vértices visitados: 4 ($A, B, C, D$).
        \item Arestas percorridas: 3 ($(A,B), (B,C), (C,D)$).
        \item \textbf{Comprimento = 3}.
    \end{itemize}
\end{frame}

\begin{frame}{Árvore Geradora Mínima}

            \item Reconhecer as aplicações práticas da AGM em problemas de engenharia.

\end{frame}

\begin{frame}{O que é uma Árvore? (Revisão)}
    \begin{block}{Definição de Árvore}
        Uma \textbf{árvore} é um grafo \textbf{conexo} e \textbf{acíclico}. Ou seja, é um grafo que:
        \begin{itemize}
            \item Possui \textbf{exatamente} um caminho entre cada par de vértices.
            \item \textbf{Não} contém ciclos.
            \item Se possui $n$ vértices, possui exatamente $n - 1$ arestas.
        \end{itemize}
    \end{block}
    \vspace{0.4cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.1, every node/.style={circle, draw, minimum size=7mm, thick}]
        \node (A) at (2,2)  {A};
        \node (B) at (0,1)  {B};
        \node (C) at (4,1)  {C};
        \node (D) at (0,0)  {D};
        \node (E) at (1.5,0){E};
        \path[thick] (A) edge (B) (A) edge (C) (B) edge (D) (B) edge (E);
    \end{tikzpicture}
    \end{center}
\end{frame}

\begin{frame}{Árvore Geradora (Spanning Tree)}
    \begin{block}{Definição}
        Dado um grafo conexo $G = (V, E)$, uma \textbf{árvore geradora} $T$ de $G$ é um \textbf{subgrafo gerador} de $G$ que é ao mesmo tempo uma \textbf{árvore}. Ou seja:
        \begin{itemize}
            \item $T$ contém \textbf{todos os vértices} de $G$ ($V_T = V$).
            \item $T$ é conexo e \textbf{não possui ciclos}.
            \item $T$ possui exatamente $|V| - 1$ arestas.
        \end{itemize}
    \end{block}
    \vspace{0.3cm}
    Um grafo pode ter \textbf{múltiplas} árvores geradoras diferentes.
\end{frame}

\begin{frame}{Árvore Geradora Mínima (AGM)}
    \begin{block}{Definição Formal}
        Dado um grafo conexo e ponderado $G = (V, E, w)$, onde $w: E \to \mathbb{R}$ atribui pesos às arestas, a \textbf{Árvore Geradora Mínima} (AGM) é uma árvore geradora $T^*$ cujo \textbf{peso total} é mínimo:
        \[
            T^* = \arg\min_{T \subseteq G,\ T \text{ é A.G.}} \sum_{e \in T} w(e)
        \]
    \end{block}
    \vspace{0.3cm}
    \begin{itemize}
        \item A AGM \textbf{conecta todos os vértices} com o menor custo total possível.
        \item Em inglês: \textit{Minimum Spanning Tree} (MST).
    \end{itemize}
\end{frame}

\subsection{Algoritmo de Kruskal}
\begin{frame}{Algoritmo de Kruskal}
    \vspace{-0.1cm}
    \begin{ttfamily}
    \textbf{KRUSKAL}($G$, $w$) \\
    \textbf{1} \hspace{0.4cm} $A \leftarrow \emptyset$ \\
    \textbf{2} \hspace{0.4cm} \textbf{para cada} vértice $v \in G.V$ \\
    \textbf{3} \hspace{0.8cm} \textsc{Criar-Conjunto}($v$) \\
    \textbf{4} \hspace{0.4cm} ordene as arestas de $G.E$ em ordem não decrescente de peso $w$ \\
    \textbf{5} \hspace{0.4cm} \textbf{para cada} aresta $(u, v) \in G.E$, na ordem não decrescente de $w$ \\
    \textbf{6} \hspace{0.8cm} \textbf{se} \textsc{Encontrar-Conjunto}($u$) $\neq$ \textsc{Encontrar-Conjunto}($v$) \\
    \textbf{7} \hspace{1.2cm} $A \leftarrow A \cup \{(u, v)\}$ \hspace{0.3cm} \textit{// não forma ciclo: adiciona} \\
    \textbf{8} \hspace{1.2cm} \textsc{Unir}($u$, $v$) \hspace{0.3cm} \textit{// une as duas componentes} \\
    \textbf{9} \hspace{0.4cm} \textbf{retorne} $A$
    \end{ttfamily}
    \vspace{0.25cm}
    \begin{block}{Estruturas auxiliares}
        \begin{itemize}
            \item \textsc{Criar-Conjunto}$(v)$: inicializa cada vértice como sua própria componente.
            \item \textsc{Encontrar-Conjunto}$(u)$: retorna o representante da componente de $u$.
            \item \textsc{Unir}$(u,v)$: funde as componentes de $u$ e $v$ (\textit{Union-Find}).
        \end{itemize}
    \end{block}
\end{frame}

\subsection{Algoritmo de Prim}
\begin{frame}{Algoritmo de Prim}
    \begin{block}{Estratégia: Greedy por Vértices}
        O algoritmo de \textbf{Prim} constrói a AGM expandindo \textbf{um vértice por vez}. A cada passo, escolhe a aresta de menor peso que conecta a árvore atual a um vértice ainda \textbf{não visitado}.
    \end{block}
    \vspace{0.3cm} 
    \textbf{Passos do algoritmo:}
    \begin{enumerate}
        \item Escolha qualquer vértice inicial. Marque-o como visitado.
        \item Considere todas as arestas que partem dos vértices visitados.
        \item Selecione a aresta de \textbf{menor peso} que chega a um vértice \textbf{não visitado}.
        \item Marque o novo vértice como visitado e repita até incluir todos.
    \end{enumerate}
\end{frame}

\begin{frame}{Comparativo: Kruskal $\times$ Prim}
    \begin{center}
    \begin{tabular}{|l|l|l|}
        \hline
        \textbf{Característica} & \textbf{Kruskal} & \textbf{Prim} \\
        \hline
        Estratégia & Por arestas & Por vértices \\
        \hline
        Estrutura auxiliar & Union-Find & Fila de prioridade \\
        \hline
        Complexidade & $\mathcal{O}(E \log E)$ & $\mathcal{O}(E \log V)$ \\
        \hline
        Ideal para & Grafos esparsos & Grafos densos \\
        \hline
        Resultado & Mesmo (AGM) & Mesmo (AGM) \\
        \hline
    \end{tabular}
    \end{center}
    \vspace{0.4cm}
    \begin{itemize}
        \item Ambos os algoritmos são \textbf{gulosos} (\textit{greedy}) e produzem a mesma AGM.
        \item A prova de corretude baseia-se na \textbf{Propriedade do Corte} e na \textbf{Propriedade do Ciclo} da teoria dos grafos.
    \end{itemize}
\end{frame}

\begin{frame}{Aplicações da AGM}
    A Árvore Geradora Mínima aparece em \textbf{problemas reais} de engenharia e computação:
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Redes de telecomunicações}: Conectar $n$ cidades com o menor comprimento total de cabo de fibra óptica.
        \vspace{0.2cm}
        \item \textbf{Redes elétricas}: Projeto de redes de distribuição de energia com custo mínimo de instalação.
        \vspace{0.2cm}
        \item \textbf{Cluster em Machine Learning}: O algoritmo \textit{single-linkage clustering} utiliza a AGM para agrupar dados similares.
        \vspace{0.2cm}
        \item \textbf{Roteamento de redes}: Protocolos como o \textbf{OSPF} usam conceitos de árvore geradora para evitar loops e otimizar o tráfego.
        \vspace{0.2cm}
        \item \textbf{Circuitos impressos (PCB)}: Conectar pinos de um circuito com o menor comprimento total de trilhas.
    \end{itemize}
\end{frame}

\section{Ciclos e PCV}
\subsection{Definição de Ciclos}
\begin{frame}{O que é um Ciclo?}
    \begin{block}{Definição de Ciclo}
        Um \textbf{ciclo} é simplesmente um \textbf{caminho fechado}. Ou seja, é um caminho que começa e termina no mesmo vértice.
    \end{block}
    \vspace{0.5cm}
    \begin{itemize}
        \item O tamanho (comprimento) do ciclo também é contado pelo número de arestas.
        \item Pelo menos 3 vértices são necessários para formar um ciclo em grafos simples não-direcionados.
    \end{itemize}
\end{frame}

\begin{frame}{Exemplo Visual: Ciclo}
    \textbf{Exemplo:} O ciclo $A \rightarrow B \rightarrow C \rightarrow A$
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, node distance=2cm, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (A) at (0,0) {A};
        \node (B) at (2,0) {B};
        \node (C) at (1,1.5) {C};
        \node (D) at (3,1) {D};
        
        \path[thick] 
        (A) edge[->, line width=1.5pt, blue, bend right=15] (B)
        (B) edge[->, line width=1.5pt, blue, bend right=15] (C)
        (C) edge[->, line width=1.5pt, blue, bend right=15] (A)
        (B) edge (D)
        (C) edge (D);
    \end{tikzpicture}
    \end{center}
    \vspace{0.3cm}
    As setas em \textbf{azul} mostram o caminho fechado (Comprimento = 3).
\end{frame}

\begin{frame}{Ciclos Simples}
    \begin{block}{Definição de Ciclo Simples}
        Um ciclo é dito \textbf{simples} se ele \textbf{não repete vértices} e não repete arestas ao longo de seu percurso, exceto obviamente pelo vértice de início e fim.
    \end{block}
    \vspace{0.5cm}
    \begin{itemize}
        \item $A \rightarrow B \rightarrow C \rightarrow A$ (Simples).
        \item $A \rightarrow B \rightarrow C \rightarrow B \rightarrow D \rightarrow A$ (\textbf{Não} é simples, pois o vértice $B$ foi visitado duas vezes).
    \end{itemize}
\end{frame}

\subsection{Ciclos Hamiltonianos e PCV}
\begin{frame}{Ciclo Hamiltoniano}
    \begin{block}{Definição de Ciclo Hamiltoniano}
        Um \textbf{ciclo Hamiltoniano} é um ciclo especial e altamente restritivo: ele é um ciclo que passa por \textbf{TODOS} os vértices do grafo \textbf{exatamente uma vez} e retorna ao vértice de origem.
    \end{block}
    \vspace{0.5cm}
    \begin{itemize}
        \item Leva o nome do matemático Sir William Rowan Hamilton.
        \item Diferente do \textit{Ciclo Euleriano} (que precisa passar por todas as \textbf{arestas}), o Hamiltoniano foca em visitar todos os \textbf{vértices}.
    \end{itemize}
\end{frame}

\begin{frame}{As Regras do Ciclo Hamiltoniano}
    Para ser considerado um Ciclo Hamiltoniano, as seguintes regras são absolutas:
    \vspace{0.5cm}
    \begin{enumerate}
        \item Tem que ser um ciclo (começa e termina no mesmo lugar).
        \item Tem que visitar $100\%$ dos vértices disponíveis no grafo.
        \item \textbf{Nenhum} vértice pode ser visitado mais de uma vez (com exceção da repetição obrigatória do vértice de origem no final).
    \end{enumerate}
\end{frame}

\begin{frame}{Exemplo: Grafo para Busca Hamiltoniana}
    Dado o grafo abaixo, existe um ciclo Hamiltoniano?
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (A) at (0,2) {A};
        \node (B) at (2,2) {B};
        \node (C) at (2,0) {C};
        \node (D) at (0,0) {D};
        
        \path[thick] 
        (A) edge (B)
        (B) edge (C)
        (C) edge (D)
        (D) edge (A)
        (A) edge (C); % Aresta diagonal extra
    \end{tikzpicture}
    \end{center}
    \vspace{0.3cm}
    Precisamos encontrar uma rota que passe por A, B, C e D apenas uma vez e volte ao início.
\end{frame}

\begin{frame}{O Ciclo Hamiltoniano Destacado}
    Sim, existe! A sequência $A \rightarrow B \rightarrow C \rightarrow D \rightarrow A$ forma um ciclo Hamiltoniano.
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.5, every node/.style={circle, draw, minimum size=8mm, thick}]
        \node (A) at (0,2) {A};

        (D) edge[->, line width=1.5pt, green!60!black] (A)

    \end{tikzpicture}
    \end{center}
    \vspace{0.3cm}
    Note que a aresta $\{A, C\}$ \textbf{não} foi utilizada. Em um ciclo Hamiltoniano, não precisamos percorrer todas as arestas, apenas todos os vértices.
\end{frame}

\begin{frame}{O Problema do Caixeiro Viajante (PCV)}
    O conceito de ciclo Hamiltoniano é a base teórica de um dos problemas mais famosos da computação: \textbf{O Problema do Caixeiro Viajante (PCV)}.
    \vspace{0.5cm}
    \begin{block}{O Problema}
        "Dada uma lista de cidades e a distância entre cada par delas, qual é a rota mais curta possível que visita cada cidade \textbf{exatamente uma vez} e retorna à cidade de origem?"
    \end{block}
\end{frame}

\begin{frame}{Relacionamento PCV e Ciclo Hamiltoniano}
    Como esses dois conceitos se ligam?
    \vspace{0.5cm}
    \begin{itemize}
        \item O Caixeiro Viajante está buscando um \textbf{Ciclo Hamiltoniano}.
        \item No PCV, o grafo possui pesos nas arestas (distâncias, custos, pedágios).
        \item O objetivo não é encontrar \textit{qualquer} ciclo Hamiltoniano (pois em um grafo completo existem muitos), mas sim encontrar o ciclo Hamiltoniano com a \textbf{menor soma de pesos} (a rota mais barata/curta).
    \end{itemize}
\end{frame}

\begin{frame}{Complexidade Computacional}
    Por que o PCV e a busca por Ciclos Hamiltonianos são tão estudados?
    \vspace{0.5cm}
    \begin{block}{Classe NP-Completo}
        A determinação da existência de um ciclo Hamiltoniano em um grafo qualquer é um problema \textbf{NP-Completo}.
    \end{block}
    \vspace{0.5cm}
    Isso significa que, assim como o problema do isomorfismo que vimos antes, \textbf{não existe} um algoritmo rápido (em tempo polinomial) conhecido para resolvê-lo garantidamente em grafos grandes.
\end{frame}

\begin{frame}{Resumo: O Desafio do Tempo de Execução}
    \begin{itemize}
        \item À medida que o número de vértices (cidades) cresce, o tempo necessário para testar todas as combinações no PCV explode combinatoriamente (Complexidade fatorial $\mathcal{O}(n!)$ na busca por força bruta).
        \vspace{0.3cm}
        \item Para resolver problemas NP-Completos na vida real (como roteamento de caminhões dos Correios, design de microchips ou logística da Uber), usamos \textbf{heurísticas} e \textbf{algoritmos de aproximação} (ex: Algoritmos Genéticos), que encontram rotas "boas o suficiente" em tempo viável, mas não garantem a perfeição matemática.
    \end{itemize}
\end{frame}

\begin{frame}{Algoritmo: Busca de Ciclo Hamiltoniano}
    Abordagem por \textit{Backtracking} (Tentativa e Erro):\\
    \vspace{0.2cm}
    \begin{ttfamily}
    
    \textbf{Função} EncontrarCiclo(caminho, vertice\_atual): \\
    \hspace{0.4cm} \textbf{Se} caminho.tamanho == Total\_de\_Vertices: \\
    \hspace{0.8cm} \textbf{Se} vertice\_atual tem aresta para caminho[0]: \\
    \hspace{1.2cm} Adicionar caminho[0] ao caminho \\
    \hspace{1.2cm} \textbf{Retornar} Verdadeiro \hspace{0.2cm}\textit{// Ciclo encontrado!} \\
    \hspace{0.8cm} \textbf{Senão}: \\
    \hspace{1.2cm} \textbf{Retornar} Falso \\
    \vspace{0.2cm}
    \hspace{0.4cm} \textbf{Para cada} vizinho de vertice\_atual: \\
    \hspace{0.8cm} \textbf{Se} vizinho não está no caminho: \\
    \hspace{1.2cm} Adicionar vizinho ao caminho \\
    \hspace{1.2cm} \textbf{Se} EncontrarCiclo(caminho, vizinho) == Verdadeiro: \\
    \hspace{1.6cm} \textbf{Retornar} Verdadeiro \\
    \hspace{1.2cm} Remover vizinho do caminho \hspace{0.2cm}\textit{// BACKTRACKING} \\
    \vspace{0.2cm}
    \hspace{0.4cm} \textbf{Retornar} Falso \hspace{0.2cm}\textit{// Nenhum ciclo encontrado}
    \end{ttfamily}
\end{frame}
