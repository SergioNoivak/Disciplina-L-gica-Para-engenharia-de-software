\item Um administrador de TI...
\begin{enumerate}[label=\alph*)]
    \item Defina formalmente o que é um \textbf{subgrafo} e um \textbf{grafo gerador} (ou subgrafo gerador).
    \item Desenhe um subgrafo de $G$ que contenha exatamente 3 vértices e 3 arestas. Ele forma um ciclo?
    \item Desenhe um grafo gerador de $G$ que seja uma árvore (ou seja, sem ciclos).
\end{enumerate}

\quadro{
\textbf{a)} \textbf{Subgrafo}: Um grafo $H = (V', E')$ é subgrafo de $G = (V, E)$ se $V' \subseteq V$ e $E' \subseteq E$. \textbf{Grafo gerador}: É um subgrafo $H$ onde o conjunto de vértices é idêntico ao original ($V' = V$), mas com menos arestas ($E' \subseteq E$). \\
\textbf{b)} Exemplo: vértices $\{1, 2, 5\}$ e arestas $\{(1,2), (2,5), (1,5)\}$. Sim, ele forma um ciclo triangular.
\begin{center}
\begin{tikzpicture}[node/.style={circle, draw=black, fill=gray!20, thick, minimum size=0.9cm}]
    \node[node] (1) at (0,2) {1}; \node[node] (2) at (2,4) {2}; \node[node] (5) at (2,2) {5};
    \draw[thick] (1) -- (2); \draw[thick] (1) -- (5); \draw[thick] (2) -- (5);
\end{tikzpicture}
\end{center}
\textbf{c)} Grafo gerador acíclico (uma árvore contendo todos os 5 vértices e 4 arestas):
\begin{center}
\begin{tikzpicture}[node/.style={circle, draw=black, fill=gray!20, thick, minimum size=0.9cm}]
    \node[node] (1) at (0,2) {1}; \node[node] (2) at (2,4) {2}; \node[node] (3) at (4,2) {3};
    \node[node] (4) at (2,0) {4}; \node[node] (5) at (2,2) {5};
    \draw[thick] (1) -- (2); \draw[thick] (2) -- (3); \draw[thick] (1) -- (4); \draw[thick] (4) -- (5);
\end{tikzpicture}
\end{center}
}

% Questão 4

\item Aplicativos de mobilidade urbana...
\begin{enumerate}[label=\alph*)]
    \item Qual a diferença fundamental entre um \textbf{caminho} e um \textbf{caminho simples}?
    \item O que caracteriza um \textbf{ciclo} em um grafo não direcionado?
    \item É possível haver um ciclo em uma árvore? Justifique com base na definição de árvore livre.
\end{enumerate}

\quadro{
\textbf{a)} Um \textbf{caminho} é qualquer sequência de arestas conectadas. Um \textbf{caminho simples} é um caminho que nunca repete vértices. \\
\textbf{b)} Um \textbf{ciclo} é um caminho simples de comprimento pelo menos 3 que começa e termina no mesmo vértice (forma um laço fechado sem repetir outros vértices intermediários). \\
\textbf{c)} \textbf{Não}. Por definição matemática, uma árvore livre é um grafo conexo e \textbf{acíclico}. Qualquer adição de aresta que crie um ciclo descaracteriza a estrutura como árvore.
}
