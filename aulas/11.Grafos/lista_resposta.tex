    \item Determine o domínio de definição das seguintes funções reais, indicando as restrições necessárias em cada caso:
    \begin{enumerate}[label=\alph*)]
        \item $f(x) = \dfrac{x + 5}{x - 7}$
        \item $g(x) = \sqrt{2x - 10}$
        \item $h(x) = \ln(x^2 - 16)$
        \item $i(x) = \dfrac{\sqrt{x + 2}}{x - 3}$
        \item $j(x) = \sqrt[3]{x - 8}$
        \item $k(x) = \dfrac{1}{x^2 - 5x + 6}$
        \item $l(x) = \log(x - 2) + \sqrt{10 - x}$
        \item $m(x) = \dfrac{1}{\sqrt{x^2 - 9}}$
        \item $n(x) = \sqrt{\dfrac{x-1}{x+2}}$
    \end{enumerate}

    \begin{solucao}
        \begin{enumerate}[label=\alph*)]
            \item $x - 7 \neq 0 \implies x \neq 7$. Portanto, $D = \mathbb{R} \setminus \{7\}$.
            \item $2x - 10 \geq 0 \implies 2x \geq 10 \implies x \geq 5$. Portanto, $D = [5, \infty)$.
            \item $x^2 - 16 > 0 \implies x^2 > 16 \implies |x| > 4$. Portanto, $D = (-\infty, -4) \cup (4, \infty)$.
            \item Restrições: $x + 2 \geq 0 \implies x \geq -2$ e $x - 3 \neq 0 \implies x \neq 3$. Portanto, $D = [-2, 3) \cup (3, \infty)$.
            \item Raiz cúbica não possui restrição para números reais. Portanto, $D = \mathbb{R}$.
            \item $x^2 - 5x + 6 \neq 0 \implies (x-2)(x-3) \neq 0 \implies x \neq 2$ e $x \neq 3$. Portanto, $D = \mathbb{R} \setminus \{2, 3\}$.
            \item Restrições: $x - 2 > 0 \implies x > 2$ e $10 - x \geq 0 \implies x \leq 10$. Portanto, $D = (2, 10]$.
            \item $x^2 - 9 > 0 \implies x^2 > 9 \implies |x| > 3$. Portanto, $D = (-\infty, -3) \cup (3, \infty)$.
            \item $\dfrac{x-1}{x+2} \geq 0$ e $x \neq -2$. Fazendo o estudo do sinal: $x < -2$ (positivo), $-2 < x < 1$ (negativo), $x \geq 1$ (positivo). Portanto, $D = (-\infty, -2) \cup [1, \infty)$.
        \end{enumerate}
    \end{solucao}

    \item Dadas as funções $f(x) = 2x - 3$ e $g(x) = x^2 + 4$, determine as expressões para as funções compostas abaixo e simplifique o resultado:
    \begin{enumerate}[label=\alph*)]
        \item $(f \circ g)(x)$
        \item $(g \circ f)(x)$
        \item $(f \circ f)(x)$
        \item Calcule o valor numérico de $(f \circ g)(2)$.
    \end{enumerate}

    \begin{solucao}
        \begin{enumerate}[label=\alph*)]
            \item $(f \circ g)(x) = f(g(x)) = 2(x^2 + 4) - 3 = 2x^2 + 8 - 3 = 2x^2 + 5$.
            \item $(g \circ f)(x) = g(f(x)) = (2x - 3)^2 + 4 = 4x^2 - 12x + 9 + 4 = 4x^2 - 12x + 13$.
            \item $(f \circ f)(x) = f(f(x)) = 2(2x - 3) - 3 = 4x - 6 - 3 = 4x - 9$.
            \item $(f \circ g)(2) = 2(2^2) + 5 = 2(4) + 5 = 8 + 5 = 13$.
        \end{enumerate}
    \end{solucao}

    \item Analise as funções abaixo e classifique-as em Injetora, Sobrejetora ou Bijetora. Justifique sua resposta analisando o domínio ($D$) e o contradomínio ($CD$).
    \begin{enumerate}[label=\alph*)]
        \item $f: \mathbb{R} \to \mathbb{R}, \quad f(x) = 3x + 5$
        \item $g: \mathbb{R} \to \mathbb{R}, \quad g(x) = x^2$
        \item $h: \mathbb{R} \to [0, \infty), \quad h(x) = x^2$
        \item $i: [0, \infty) \to [0, \infty), \quad i(x) = x^2$
        \item $j: \mathbb{R} \to \mathbb{R}, \quad j(x) = x^3$
        \item $k: \mathbb{R} \to \mathbb{R}, \quad k(x) = |x|$
        \item $l: \mathbb{R} \to (0, \infty), \quad l(x) = e^x$
        \item $m: (0, \infty) \to \mathbb{R}, \quad m(x) = \ln(x)$
        \item $n: \mathbb{R} \setminus \{0\} \to \mathbb{R} \setminus \{0\}, \quad n(x) = \frac{1}{x}$
        \item $p: \mathbb{R} \to \mathbb{R}, \quad p(x) = \sin(x)$
    \end{enumerate}

    \begin{solucao}
        \begin{enumerate}[label=\alph*)]
            \item \textbf{Bijetora}. É uma função linear não constante; cada $x$ leva a um $y$ único (injetora) e a imagem cobre todo o contradomínio $\mathbb{R}$ (sobrejetora).
            \item \textbf{Nem injetora, nem sobrejetora}. Não é injetora pois $g(1) = g(-1) = 1$. Não é sobrejetora pois a imagem é $[0, \infty) \neq \mathbb{R}$ (não há $x$ real tal que $x^2 = -1$).
            \item \textbf{Sobrejetora}. Não é injetora (mesmo motivo acima), mas é sobrejetora pois o contradomínio foi restrito à imagem da função $[0, \infty)$.
            \item \textbf{Bijetora}. Com o domínio restrito a $[0, \infty)$, para cada $y \geq 0$ existe um único $x = \sqrt{y} \geq 0$.
            \item \textbf{Bijetora}. A função cubo é estritamente crescente em $\mathbb{R}$ e sua imagem é $\mathbb{R}$.
            \item \textbf{Nem injetora, nem sobrejetora}. Semelhante a $x^2$, $k(1) = k(-1) = 1$ e a imagem é $[0, \infty)$.
            \item \textbf{Bijetora}. A exponencial é estritamente crescente e sua imagem é $(0, \infty)$.
            \item \textbf{Bijetora}. O logaritmo natural é a inversa da exponencial, sendo injetora no seu domínio e sobrejetora em $\mathbb{R}$.
            \item \textbf{Bijetora}. É sua própria inversa; injetora e sobrejetora nos conjuntos definidos.
            \item \textbf{Nem injetora, nem sobrejetora}. Não é injetora (ex: $\sin(0) = \sin(\pi) = 0$) e a imagem é $[-1, 1] \neq \mathbb{R}$.
        \end{enumerate}
    \end{solucao}

    \item Utilize o Teste da Reta Vertical e o Teste da Reta Horizontal para classificar as curvas representadas nos gráficos abaixo como: (I) Função ou não-função; (II) Injetora ou não-injetora.

    \begin{solucao}
        \begin{itemize}

            \item \textbf{Gráfico B (Parábola lateral):} (I) Não-função (reta vertical toca em dois pontos).
            \item \textbf{Gráfico C (Reta inclinada):} (I) Função; (II) Injetora.
            \item \textbf{Gráfico D (Cúbica):} (I) Função; (II) Injetora.
            \item \textbf{Gráfico E (Círculo):} (I) Não-função.
            \item \textbf{Gráfico F (Reta horizontal):} (I) Função; (II) Não-injetora.
            \item \textbf{Gráfico G (Senoide):} (I) Função; (II) Não-injetora.
            \item \textbf{Gráfico H (Valor Absoluto):} (I) Função; (II) Não-injetora.
            \item \textbf{Gráfico I (Semicírculo):} (I) Função; (II) Não-injetora.
            \item \textbf{Gráfico J (Exponencial):} (I) Função; (II) Injetora.
            \item \textbf{Gráfico K (Logaritmo):} (I) Função; (II) Injetora.
        \end{itemize}
    \end{solucao}

    \item Considere a função $f: A \to B$ representada pelo diagrama de Venn abaixo, onde $A = \{1, 2, 3\}$ e $B = \{a, b, c, d\}$.
    
    Determine:
    \begin{enumerate}[label=\alph*)]
        \item O conjunto Domínio ($D$).
        \item O conjunto Contradomínio ($CD$).
        \item O conjunto Imagem ($Im$).
        \item Esta função é sobrejetora? Justifique.
        \item Esta função é injetora? Justifique.
    \end{enumerate}

    \begin{solucao}
        \begin{enumerate}[label=\alph*)]
            \item $D = A = \{1, 2, 3\}$.
            \item $CD = B = \{a, b, c, d\}$.
            \item $Im = \{a, c\}$. (Apenas os elementos de $B$ que recebem flechas).
            \item \textbf{Não}. Pois $Im \neq CD$. Elementos como $\{b, d\}$ pertencem ao contradomínio mas não à imagem.
            \item \textbf{Não}. Pois existem elementos distintos no domínio ($2$ e $3$) que possuem a mesma imagem ($c$). Ou seja, $f(2) = f(3) = c$.
        \end{enumerate}
    \end{solucao}

\end{enumerate}
