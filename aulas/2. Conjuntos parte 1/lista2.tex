\section*{Instruções}

\begin{block}{Utilize definições formais da lógica proposicional}
Considere proposições $P$ e $Q$ com valores lógicos em $\{V, F\}$.
A implicação lógica
\[
P \Rightarrow Q
\]
é definida como falsa apenas quando $P$ é verdadeira e $Q$ é falsa,
sendo verdadeira em todos os demais casos.
\end{block}

Justifique rigorosamente todas as respostas utilizando definições formais,
tabelas-verdade e argumentos lógicos, evitando interpretações informais.



\section*{Exercícios}

\begin{enumerate}

\item Indique quais das sentenças abaixo são proposições.
Caso sejam proposições, classifique-as como verdadeiras (V) ou falsas (F).
\begin{enumerate}[label=(\hspace{1.2em})]
    \item $4$ é um número primo.
    \item Todo algoritmo correto termina.
    \item Qual é o valor de $x$?
    \item $2^5 = 32$.
\end{enumerate}

\item Considere a implicação:
\[
\text{``Se um número é múltiplo de 4, então é par.''}
\]
Identifique claramente:
\begin{enumerate}[label=(\hspace{1.2em})]
    \item a hipótese;
    \item a conclusão.
\end{enumerate}

\item Construa a tabela-verdade completa da proposição
\[
P \Rightarrow Q.
\]
Em seguida, indique explicitamente em quais casos a implicação é falsa.

\item Seja a afirmação:
\[
\text{``Se um programa possui erro, então ele não funciona corretamente.''}
\]
\begin{enumerate}
    \item Defina proposições $P$ e $Q$;
    \item Analise a validade lógica da implicação utilizando tabela-verdade.
\end{enumerate}

\item Para a implicação
\[
P \Rightarrow Q,
\]
\begin{enumerate}
    \item escreva a \textbf{volta da implicação};
    \item explique por que, em geral, ela não é logicamente equivalente
    à implicação original.
\end{enumerate}

\item Verifique se a proposição abaixo é verdadeira:
\[
(x \text{ é par}) \Leftrightarrow (x^2 \text{ é par})
\]
Justifique utilizando implicações diretas e a volta da implicação.

\item Explique, com base na definição formal da implicação lógica,
por que toda proposição da forma
\[
P \Rightarrow Q
\]
é verdadeira quando $P$ é falsa.

\item Analise a proposição:
\[
(x \text{ é par}) \Rightarrow (x \text{ não é ímpar})
\]
\begin{enumerate}
    \item Construa a tabela-verdade.
    \item Explique o resultado obtido.
\end{enumerate}

\item Explique, usando a definição formal da implicação lógica,
por que a proposição
\[
(x \text{ é ímpar}) \Rightarrow (x \text{ é par})
\]
pode ser verdadeira para alguns valores de $x$,
mesmo sendo falsa como afirmação matemática geral.


  
\item Considere as seguintes proposições para números inteiros \(x\) e \(y\):
\[
P: x \text{ e } y \text{ são ímpares}, \qquad Q: x \cdot y \text{ é ímpar}.
\]
\begin{enumerate}
    \item Construa a tabela-verdade mostrando todos os casos possíveis para \(P\) e \(Q\).
    \item Analise as implicações:
    \[
    P \Rightarrow Q \quad \text{e} \quad Q \Rightarrow P.
    \]
    \item Com base nas duas implicações, conclua se a bicondicional
    \[
    P \Leftrightarrow Q
    \]
    é verdadeira ou falsa, e explique logicamente o resultado.
\end{enumerate}

\end{enumerate}
