\section*{Formulário de Referência}

\begin{tcolorbox}[colback=gray!5,colframe=black,title=Binômio e Multinômio de Newton,fonttitle=\bfseries]
    \textbf{Binômio de Newton:}
    \begin{itemize}
        \item Expansão: $(a+b)^n = \sum_{k=0}^n \binom{n}{k} a^{n-k} b^k$
        \item Termo Geral: $T_{k+1} = \binom{n}{k} a^{n-k} b^k$
        \item Coeficiente Binomial: $\binom{n}{k} = \frac{n!}{k!(n-k)!}$
        \item Soma dos Coeficientes: Substituir variáveis por 1.
    \end{itemize}
    \smallskip
    \textbf{Multinômio de Newton:}
    \begin{itemize}
        \item Termo Geral: $\frac{n!}{k_1! k_2! \dots k_m!} x_1^{k_1} x_2^{k_2} \dots x_m^{k_m}$
        \item Número de termos distintos: $N = \binom{n+m-1}{m-1}$
    \end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=gray!5,colframe=black,title=Análise Combinatória,fonttitle=\bfseries]
    \begin{itemize}[noitemsep]
        \item \textbf{Fatorial:} $n! = n \cdot (n-1) \cdot (n-2) \dots 1 \quad (0! = 1)$
        \item \textbf{Permutação Simples:} $P_n = n!$
        \item \textbf{Arranjo Simples:} $A_{n,k} = \frac{n!}{(n-k)!}$
        \item \textbf{Combinação Simples:} $C_{n,k} = \binom{n}{k} = \frac{n!}{k!(n-k)!}$
        \item \textbf{Permutação com Repetição:} $P_n^{n_1, n_2, \dots, n_k} = \frac{n!}{n_1! n_2! \dots n_k!}$
    \end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=gray!5,colframe=black,title=Propriedades da Álgebra Booleana,fonttitle=\bfseries]
    \begin{itemize}[noitemsep]
        \item \textbf{Distributiva:} $A + (B \cdot C) = (A + B) \cdot (A + C)$
        \item \textbf{De Morgan:} $\overline{A \cdot B} = \bar{A} + \bar{B} \quad \text{e} \quad \overline{A + B} = \bar{A} \cdot \bar{B}$
        \item \textbf{Absorção:} $A + (A \cdot B) = A \quad \text{e} \quad A \cdot (A + B) = A$
        \item \textbf{Complemento:} $A + \bar{A} \cdot B = A + B \quad \text{e} \quad \bar{A} + A \cdot B = \bar{A} + B$
        \item \textbf{Consenso:} $A \cdot B + \bar{A} \cdot C + B \cdot C = A \cdot B + \bar{A} \cdot C$
    \end{itemize}
\end{tcolorbox}
