    \textbf{Binômio:} $(a+b)^n = \sum_{k=0}^n \binom{n}{k} a^{n-k} b^k$ \\[0.15cm]
    \textbf{Termo Geral:} $T_{k+1} = \binom{n}{k} a^{n-k} b^k$ \\[0.15cm]
    \textbf{Permutação Simples:} $P_n = n!$ \\[0.15cm]
    \textbf{Permutação (Rep.):} $P_n^{k_1, \dots, k_m} = \frac{n!}{k_1! \dots k_m!}$ \\[0.15cm]
    \textbf{Arranjo Simples:} $A_{n,k} = \frac{n!}{(n-k)!}$ \\[0.15cm]
    \textbf{Arranjo (Rep.):} $AR_{n,k} = n^k$ \\[0.15cm]
    \textbf{Combinação Simples:} $C_{n,k} = \binom{n}{k} = \frac{n!}{k!(n-k)!}$ \\[0.15cm]
    \textbf{Combinação (Rep.):} $CR_{n,k} = \binom{n+k-1}{k}$ \\[0.15cm]
    \textbf{Multinômio:} $\frac{n!}{k_1! \dots k_m!} x_1^{k_1} \dots x_m^{k_m}$

    \noindent \textbf{4. Funções e Relações}\\
    
    Relação ($R: A \to B$): $R \subseteq A \times B$ \\[0.1cm]
    Função ($f: A \to B$): $\forall x \in A, \exists! y \in B \mid (x, y) \in f$ \\[0.1cm]
    Injetora:$\forall x_1, x_2 \in A, f(x_1) = f(x_2) \implies x_1 = x_2$ \\[0.1cm]
    Sobrejetora: $\forall y \in B, \exists x \in A \mid f(x) = y$ \\[0.1cm]
    Bijetora: $\forall x_1, x_2 \in A, f(x_1) = f(x_2) \implies x_1 = x_2$ $\land$ $\forall y \in B, \exists x \in A \mid f(x) = y$
